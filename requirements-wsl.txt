# Dependency set for the WSL2 + CUDA environment used by this project.
#
# Differs from the upstream requirements.txt, whose pins are ~1 year old and include versions
# that no longer install on current Python. Install torch FIRST from the CUDA index, then this
# file:
#
#   pip install --index-url https://download.pytorch.org/whl/cu124 torch torchvision
#   pip install -r requirements-wsl.txt
#   pip install -e .
#
# See scripts/setup_wsl.sh, which does all of the above plus the system packages.

# --- core training stack -------------------------------------------------------------------
# torch / torchvision are intentionally absent: they must come from the CUDA wheel index above,
# otherwise pip resolves the CPU-only build.
pytorch-lightning>=2.5,<3
hydra-core>=1.3,<2
hydra-zen>=0.15,<1
omegaconf>=2.3,<3

# --- SigLIP2 + GPT-2 -----------------------------------------------------------------------
transformers>=4.49          # SigLIP2 support landed in 4.49
sentencepiece
accelerate

# --- video / compressed domain -------------------------------------------------------------
# decord 0.6.0 has no wheels beyond CPython 3.10; eva-decord is the maintained fork that does.
# setup_wsl.sh tries decord first and falls back. cv_reader is NOT on PyPI - build it from
# https://github.com/yaojie-shen/Compressed-Video-Reader (see docs/implementation-plan.md 1.4).
decord==0.6.0; python_version < "3.11"
eva-decord>=0.6.1; python_version >= "3.11"
av>=12
opencv-python>=4.10,<5      # opencv 5 is very new; stay on the 4.x API this code was written for
lz4>=4

# --- evaluation ----------------------------------------------------------------------------
pycocoevalcap==1.2          # requires a JRE (default-jre) for PTBTokenizer / METEOR

# --- everything else -----------------------------------------------------------------------
numpy>=2                    # decord and fvcore both import cleanly under numpy 2
pandas>=2.2,<3              # pandas 3.0 is a breaking release; usage here is trivial, so pin low
pillow>=10
einops>=0.8
easydict>=1.13
timm>=1.0
ftfy>=6.3
regex
fvcore>=0.1.5
h5py>=3.13

terminaltables>=3.1
tabulate>=0.9
colorlog>=6.9
joblib>=1.4
tqdm>=4.67
requests>=2.32
pyyaml>=6
packaging                   # clip.py uses it directly; pkg_resources is gone in setuptools>=81
